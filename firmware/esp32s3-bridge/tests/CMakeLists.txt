cmake_minimum_required(VERSION 3.16)
project(bridge_tests C)
set(CMAKE_C_STANDARD 11)
add_executable(bridge_tests test_bridge.c ../main/bridge_core.c)
add_executable(voice_tests test_voice.c ../main/voice_core.c)
target_include_directories(voice_tests PRIVATE ../main)
target_include_directories(bridge_tests PRIVATE ../main)
if(MSVC)
    target_compile_options(bridge_tests PRIVATE /W4 /WX /UNDEBUG)
    target_compile_options(voice_tests PRIVATE /W4 /WX /UNDEBUG)
else()
    target_compile_options(bridge_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_compile_options(voice_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
enable_testing()
add_test(NAME bridge_core COMMAND bridge_tests)
add_test(NAME voice_core COMMAND voice_tests)
