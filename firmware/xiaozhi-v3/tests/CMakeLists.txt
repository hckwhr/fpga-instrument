cmake_minimum_required(VERSION 3.16)
project(display_lock_tests CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(display_lock_tests test_display_lock.cc)
target_include_directories(display_lock_tests PRIVATE stubs ../main/display)
if(MSVC)
    target_compile_options(display_lock_tests PRIVATE /W4 /WX /UNDEBUG)
else()
    target_compile_options(display_lock_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
enable_testing()
add_test(NAME display_lock COMMAND display_lock_tests)
